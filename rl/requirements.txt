#   pip install vllm==0.8.5.post1 --extra-index-url https://download.pytorch.org/whl/cu124
#   pip install transformers==4.51.3          # 5.x breaks vllm 0.8.5 import
#   pip install -e /path/to/verl@v0.8.0 --no-deps
#   # flash-attn optional; if the FSDP model demands it, either install it or
#   # pass actor_rollout_ref.model.attn_implementation=sdpa
#
# Result: verl 0.8.0.dev | vllm 0.8.5.post1 | torch 2.6.0+cu124 | transformers 4.51.3
